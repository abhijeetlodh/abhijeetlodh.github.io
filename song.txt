    // Create test cases: PRs with known issues
    private List<GoldenTestCase> _goldenDataset = new()

        { 
            Name = "SQL Injection Vulnerability",
            Code = "var query = $\"SELECT * FROM Users WHERE Id = {userId}\";",

        },
        // Add 50+ test cases

    {
        var results = new List<EvalResult>();
        
        foreach (var testCase in _goldenDataset)
        {
            var review = await _orchestrator.ReviewCode(testCase.Context);
            var truePositives = review.Findings.Count(f => testCase.ExpectedFindings.Contains(f.Title));
            var falsePositives = review.Findings.Count(f => !testCase.ExpectedFindings.Contains(f.Title));
            var falseNegatives = testCase.ExpectedFindings.Count(e => !review.Findings.Any(f => f.Title == e));
            
            results.Add(new EvalResult(truePositives, falsePositives, falseNegatives));
        }
        
        return CalculateMetrics(results); // Precision, Recall, F1
    }
}

```csharp
public class ReviewWorker : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        await foreach (var job in _messageQueue.Consume<ReviewJob>(stoppingToken))
        {
            try
            {
                // 1. Clone repository
                await CloneRepo(job.RepositoryUrl, job.HeadBranch);
                
                // 2. Build knowledge graph
                var graph = await _knowledgeGraph.BuildFromSolution("./repo");
                
                // 3. Get changed files
                var changedFiles = await GetChangedFiles(job.BaseBranch, job.HeadBranch);
                
                // 4. Build context
                var context = await _contextEngine.BuildContext(changedFiles, graph);
                
                // 5. Run review
                var review = await _orchestrator.ReviewCode(context);
                
                // 6. Run build
                review.BuildResult = await _buildExecutor.ExecuteCloneBuild("./repo", job.HeadBranch);
                
                // 7. Apply auto-fixes
                if (review.Findings.Any(f => f.CanAutoFix))
                {
                    await _autoFixer.ApplyFixes(review.Findings.Where(f => f.CanAutoFix).ToList(), "./repo");
                }
                
                // 8. Store results
                await _cache.Set(job.Id, review);
                
                // 9. Post to GitHub
                await PostReviewComment(job.PullRequestNumber, review);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Review failed for job {JobId}", job.Id);
            }
        }
    }
}

### Context Optimization Strategy

```csharp
public class SmartContextBuilder
{
    public async Task<string> BuildOptimalContext(List<string> changedFiles, ICodeKnowledgeGraph graph)
    {
        var context = new StringBuilder();
        var tokenCount = 0;
        const int MAX_TOKENS = 100_000; // Leave 28K for response
        
        // PRIORITY 1: Changed files (ALWAYS include full content)
        foreach (var file in changedFiles)
        {
            var fileContent = await File.ReadAllTextAsync(file);
            context.AppendLine($"## CHANGED FILE: {file}");
            context.AppendLine("```csharp");
            context.AppendLine(fileContent);
            context.AppendLine("```");
            tokenCount += CountTokens(fileContent);
        }
        
        // PRIORITY 2: Direct dependencies
        var dependencies = await graph.GetDirectDependencies(changedFiles);
        foreach (var dep in dependencies)
        {
            if (tokenCount + dep.TokenCount < MAX_TOKENS * 0.7)
            {
                context.AppendLine($"## DEPENDENCY: {dep.FilePath}");
                context.AppendLine("```csharp");
                context.AppendLine(dep.Content);
                context.AppendLine("```");
                tokenCount += dep.TokenCount;
            }
            else
            {
                // Summarize instead
                var summary = await SummarizeDependency(dep);
                context.AppendLine($"## DEPENDENCY (SUMMARY): {dep.FilePath}");
                context.AppendLine(summary);
                tokenCount += CountTokens(summary);
            }
        }
        
        // PRIORITY 3: Method signatures of callers
        var callers = await graph.GetCallingMethods(changedFiles);
        context.AppendLine("## METHODS THAT CALL THIS CODE:");
        foreach (var caller in callers)
        {
            context.AppendLine($"- {caller.Signature} in {caller.FilePath}");
        }
        
        // PRIORITY 4: Test files
        var tests = await graph.GetRelatedTests(changedFiles);
        if (tokenCount < MAX_TOKENS * 0.8)
        {
            context.AppendLine("## RELATED TESTS:");
            foreach (var test in tests)
            {
                context.AppendLine("```csharp");
                context.AppendLine(test.Content);
                context.AppendLine("```");
            }
        }
        
        return context.ToString();
    }
}
```

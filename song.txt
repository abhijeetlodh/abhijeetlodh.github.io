Write-Host "`n===== 3. CONFIGURATION FILES =====" -ForegroundColor Cyan
$files = @(
    "$HOME\.npmrc",
    "$PWD\.npmrc",
    (npm config get globalconfig)
)

foreach ($file in $files) {
    if (Test-Path $file) {
        Write-Host "Found: $file"
        Select-String -Path $file `
          -Pattern '^\s*(registry|@wdio:registry|cafile|strict-ssl)\s*=' |
          ForEach-Object {
              $_.Line -replace '://[^/\s]+@','://[REDACTED]@'
          }
    }
}

Write-Host "`n===== 7. NETWORK CONNECTIVITY =====" -ForegroundColor Cyan
$registry = (npm config get registry).Trim()

try {
    $registryHost = ([uri]$registry).Host
    Resolve-DnsName $registryHost -ErrorAction Continue
    Test-NetConnection $registryHost -Port 443
} catch {
    Write-Host $_.Exception.Message
}
